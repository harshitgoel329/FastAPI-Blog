# this is a test file for a synchronous FastAPI app, which is not the main app of the project. It is used to test the FastAPI TestClient functionality.

from fastapi import FastAPI
from fastapi.testclient import TestClient


demo_app = FastAPI()

@demo_app.get("/")
def demo_home():
    return {"message": "Hello"}

client = TestClient(demo_app)

def test_homepage():
    response = client.get("/")
    assert response.status_code == 200
